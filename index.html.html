// CAPTURA Y CREACIÓN DE PIEZAS function takePuzzleSnapshot() { const video = document.getElementById('puzzleVideo'); const canvas = document.createElement('canvas'); // Aumentamos la resolución a 600x360 (relación 5:3 exacta para 15 piezas) // Esto hace que las piezas sean mucho más nítidas al agrandar la pantalla canvas.width = 600; canvas.height = 360; const ctx = canvas.getContext('2d'); ctx.translate(canvas.width, 0); ctx.scale(-1, 1); // Dibujamos el video en el canvas ctx.drawImage(video, 0, 0, canvas.width, canvas.height); stopPuzzleAiCamera(); // Subimos un poquito la calidad de compresión (0.7 en lugar de 0.6) const base64Img = canvas.toDataURL('image/jpeg', 0.7); generatePuzzleArrayAndSync(base64Img); }
